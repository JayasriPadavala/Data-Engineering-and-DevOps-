31. **Check if a given number is prime**  
    - Implement using loops and conditions.

#!/bin/bash
is_prime() {
  num=$1
  if [ $num -le 1 ]; then
    echo "$num is not a prime number."
    return
  fi
  for (( i=2; i<num; i++ ))
  do
    if [ $((num % i)) -eq 0 ]; then
      echo "$num is not a prime number."
      return
    fi
  done
  echo "$num is a prime number."
}
echo "Enter a number:"
read number
is_prime $number
